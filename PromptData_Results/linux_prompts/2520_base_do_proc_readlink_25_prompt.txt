	char *tmp = (char *)__get_free_page(GFP_KERNEL);

	if (IS_ERR(pathname))

	len = tmp + PAGE_SIZE - 1 - pathname;

	if (len > buflen)
		len = buflen;
	if (copy_to_user(buffer, pathname, len))
		len = -EFAULT;
 out:

	return len;
}
