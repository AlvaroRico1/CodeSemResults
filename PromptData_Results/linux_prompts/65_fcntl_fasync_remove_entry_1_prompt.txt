	struct fasync_struct *fa, **fp;

	for (fp = fapp; (fa = *fp) != NULL; fp = &fa->fa_next) {

		filp->f_flags &= ~FASYNC;
		result = 1;
		break;
	}
	spin_unlock(&fasync_lock);
	spin_unlock(&filp->f_lock);
	return result;
}
