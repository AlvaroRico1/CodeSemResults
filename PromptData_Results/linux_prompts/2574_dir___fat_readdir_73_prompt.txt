	if (isvfat && !short_only) {
		if (de->name[0] == DELETED_FLAG)
			goto record_end;
		if (de->attr != ATTR_EXT && (de->attr & ATTR_VOLUME))

		else if (status == PARSE_EOF)

	if (fake_offset && ctx->pos < 2)
		ctx->pos = 2;

	if (!memcmp(de->name, MSDOS_DOT, MSDOS_NAME)) {

record_end:

	ctx->pos = cpos;
	goto get_new;

	else

out:
	mutex_unlock(&sbi->s_lock);

	return ret;
}
