	if (test_bit(NFS_IOHDR_ERROR, &hdr->flags))

			set_page_dirty(page);
		bytes += req->wb_bytes;
		nfs_list_remove_request(req);
		nfs_release_request(req);
	}

	if (put_dreq(dreq))
		nfs_direct_complete(dreq);

}
