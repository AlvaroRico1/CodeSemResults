	struct nfs4_mount_data *data = (struct nfs4_mount_data *)options;
	char *c;

			return -EFAULT;

		args->nfs_server.port = ntohs(((struct sockaddr_in *)sap)->sin_port);

		if (data->auth_flavourlen) {
