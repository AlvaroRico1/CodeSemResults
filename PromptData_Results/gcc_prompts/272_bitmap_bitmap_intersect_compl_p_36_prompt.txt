{
  const bitmap_element *a_elt;
  const bitmap_element *b_elt;

  for (a_elt = a->first, b_elt = b->first;

    {

      else if (b_elt->indx < a_elt->indx)
	b_elt = b_elt->next;

	    if (a_elt->bits[ix] & ~b_elt->bits[ix])
	      return true;
	  a_elt = a_elt->next;
	  b_elt = b_elt->next;
	}
    }
  return a_elt != NULL;
}
