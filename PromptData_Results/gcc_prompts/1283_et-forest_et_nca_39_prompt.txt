  et_splay (o1);
  l = o1->prev;
  r = o1->next;
  if (l)

  et_splay (o2);

  if (l == o2 || (l && l->parent != NULL))

      set_prev (o1, o2);

    }
  else if (r == o2 || (r && r->parent != NULL))

      ret = o2->prev;

      set_next (o1, o2);
      if (l)
	l->parent = o1;
    }
  else

      if (l)
	l->parent = o1;

	r->parent = o1;

      mn = o1->depth;

      om = o2;
      mn = o2->depth + o1->depth;

  et_check_tree_sanity (o2);
#endif

  if (ret && ret->min + o1->depth + o2->depth < mn)
    return ret->min_occ->of;
  else

}
