	u_int			 leading_width, copy_width;

			if (width <= skip) {

			} else
				copy_width = leading_width;
			if (copy_width != 0) {
				if (n == 1)

				else {
					memset(out, '#', 2 * copy_width);

				}

			}

			if (width >= skip)
				*out++ = *cp;
			width++;
			cp++;
		} else
			cp++;
	}
	*out = '\0';
	return (copy);
}
