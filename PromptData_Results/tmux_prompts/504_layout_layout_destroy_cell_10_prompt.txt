	if (lcparent == NULL) {

	/* Merge the space into the previous or next cell. */
	if (lc == TAILQ_FIRST(&lcparent->cells))

			lc->xoff = 0; lc->yoff = 0;
			*lcroot = lc;
		} else

	}
}
