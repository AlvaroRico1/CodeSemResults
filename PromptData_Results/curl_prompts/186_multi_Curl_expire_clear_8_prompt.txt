    while(list->size > 0) {
      Curl_llist_remove(list, list->tail, NULL);
    }

    infof(data, "Expire cleared (transfer %p)", data);
#endif
    nowp->tv_sec = 0;
    nowp->tv_usec = 0;
  }
}
