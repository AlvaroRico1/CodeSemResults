{
  curl_off_t size;
  size_t boundarysize;

  boundarysize = 4 + strlen(mime->boundary) + 2;
  size = boundarysize;  /* Final boundary - CRLF after headers. */

    if(sz < 0)

    if(size >= 0)
      size += boundarysize + sz;
  }

  return size;
}
