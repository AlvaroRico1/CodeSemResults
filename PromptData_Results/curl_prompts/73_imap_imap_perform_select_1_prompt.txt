  struct imap_conn *imapc = &conn->proto.imapc;

  if(!mailbox)

  /* Send the SELECT command */
  result = imap_sendf(data, conn, "SELECT %s", mailbox);

  free(mailbox);

  if(!result)
    state(data, IMAP_SELECT);

  return result;
}
