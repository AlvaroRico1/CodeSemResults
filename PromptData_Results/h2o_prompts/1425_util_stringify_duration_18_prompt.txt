size_t stringify_duration(char *buf, int64_t usec)
{
    int32_t msec = (int32_t)(usec / 1000);
    usec -= ((int64_t)msec * 1000);
    char *pos = buf;
    pos += sprintf(pos, "dur=%" PRId32, msec);
    if (usec != 0) {
        *pos++ = '.';
        int denom;
        for (denom = 100; denom != 0; denom /= 10) {
            int d = (int)usec / denom;
            *pos++ = '0' + d;
            usec -= d * denom;
            if (usec == 0)
                break;
        }
    }
    return pos - buf;
}
