  while (max_part_id > min_part_id) {

    if (range_array[loc_part_id] <= part_func_value)
      min_part_id = loc_part_id + 1;

      !part_info->defined_max_value)

  DBUG_PRINT("exit", ("partition: %d", *part_id));

}
