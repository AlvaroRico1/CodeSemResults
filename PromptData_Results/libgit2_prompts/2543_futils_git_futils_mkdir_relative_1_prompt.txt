				error = -1;
				goto done;
			}

			opts->perfdata.mkdir_calls++;

				error = -1;
				goto done;
			}

				goto done;
		}

		if ((error = mkdir_validate_mode(

			goto done;

		if (opts->dir_map && opts->pool) {

			size_t alloc_size;

			error = GIT_ENOTFOUND;
		}

	return error;
}
