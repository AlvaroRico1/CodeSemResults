static long diff_context_find(
	const char *line,
	long line_len,
	char *out,
	long out_size,
	void *payload)

	if (git_str_set(&ctxt->line, line, (size_t)line_len) < 0)

	if (out_size > (long)ctxt->line.size)
		out_size = (long)ctxt->line.size;
	memcpy(out, ctxt->line.ptr, (size_t)out_size);

	return out_size;
}
