		 * "diff3 -m" output does not make sense for anything
		 * more aggressive than XDL_MERGE_EAGER.

		if (XDL_MERGE_EAGER < level)

		i2 = xscr2->i1 + xscr2->chg1;

			return -1;
		}
		result->size = size;

				      ancestor_name, favor, changes,
				      result->ptr, style, marker_size);
	}

}
