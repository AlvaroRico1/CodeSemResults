	const char *new,

	git_apply_hunk_cb hunk_cb,
	void *payload)

		cl_assert_equal_i(0100644, mode);
	}

	git__free(filename);
	git_str_dispose(&result);
	git_str_dispose(&patchbuf);
	git_patch_free(patch);

	return error;
}
