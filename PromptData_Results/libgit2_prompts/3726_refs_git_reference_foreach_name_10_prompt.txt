	git_reference_foreach_name_cb callback,
	void *payload)
{
	git_reference_iterator *iter;

	int error;

			break;
		}
	}

	if (error == GIT_ITEROVER)
		error = 0;

	git_reference_iterator_free(iter);
	return error;
}
